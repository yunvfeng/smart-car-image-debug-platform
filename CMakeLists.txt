cmake_minimum_required(VERSION 3.20)
project(VisionLab LANGUAGES C)
set(SMART_CAR_DIR "${CMAKE_CURRENT_SOURCE_DIR}/smart-car" CACHE PATH "Original firmware checkout")
set(VISION "${SMART_CAR_DIR}/project/code")
set(VISIONLAB_IMAGE_SOURCE "${VISION}/image.c" CACHE FILEPATH "Selected editable algorithm entry")
if(NOT EXISTS "${VISIONLAB_IMAGE_SOURCE}")
    message(FATAL_ERROR "Select an existing VISIONLAB_IMAGE_SOURCE")
endif()
get_filename_component(ALGORITHM_DIR "${VISIONLAB_IMAGE_SOURCE}" DIRECTORY)
get_filename_component(SELECTED_NAME "${VISIONLAB_IMAGE_SOURCE}" NAME)
set(MODULE_NAMES image image2 ring visual_avoidance laser)
get_filename_component(SELECTED_MODULE "${SELECTED_NAME}" NAME_WE)
if(NOT SELECTED_MODULE IN_LIST MODULE_NAMES)
    set(SELECTED_MODULE image)
endif()
set(VISION_MODULES)
foreach(MODULE IN LISTS MODULE_NAMES)
    if(MODULE STREQUAL SELECTED_MODULE)
        list(APPEND VISION_MODULES "${VISIONLAB_IMAGE_SOURCE}")
    elseif(EXISTS "${ALGORITHM_DIR}/${MODULE}.c")
        list(APPEND VISION_MODULES "${ALGORITHM_DIR}/${MODULE}.c")
    else()
        list(APPEND VISION_MODULES "${VISION}/${MODULE}.c")
    endif()
endforeach()
add_executable(visionlab_replay
    visionlab/c/replay.c visionlab/c/platform.c
    ${VISION_MODULES})
target_include_directories(visionlab_replay PRIVATE visionlab/c/platform "${ALGORITHM_DIR}" "${VISION}")
set_target_properties(visionlab_replay PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(visionlab_replay PRIVATE -g3 -O0 -Wall -Wextra -Wpedantic)
    if(VISIONLAB_SOURCE_ROOT)
        # A UTF-8 response file avoids cmd.exe's ANSI conversion of Chinese paths.
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/debug-map.rsp"
            "\"-fdebug-prefix-map=${VISIONLAB_SOURCE_ROOT}=.\"\n")
        target_compile_options(visionlab_replay PRIVATE "@${CMAKE_CURRENT_BINARY_DIR}/debug-map.rsp")
    endif()
endif()
